pipeline {
  agent any
  environment { ANSIBLE_HOST_KEY_CHECKING = 'False' }

  stages {
    stage('Checkout main') {
      steps { git branch: 'main', url: 'https://github.com/<your-username>/aws-capstone-exam.git' }
    }

    stage('Deploy v1 (Ansible)') {
      steps {
        sh '''
          sed -i 's/^app_version:.*/app_version: "v1"/' ansible/group_vars/web.yml
          ansible-playbook -i ansible/hosts ansible/site.yml
        '''
      }
    }

    stage('Deploy v2 (on next push)') {
      steps {
        sh '''
          sed -i 's/^app_version:.*/app_version: "v2"/' ansible/group_vars/web.yml
          ansible-playbook -i ansible/hosts ansible/site.yml
        '''
      }
    }
  }
}
